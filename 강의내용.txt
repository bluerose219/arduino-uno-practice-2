(강사님코드)  pot_dimmer.ino
const byte POT_PIN = A0;
const byte LED_PIN = 9;  // Uno에서 ~ 표시가 있는 PWM 핀
unsigned long lastReport = 0;

void setup() {
  Serial.begin(9600);
  pinMode(LED_PIN, OUTPUT);
  Serial.println(F("Potentiometer ready: ADC 0..1023 -> PWM 0..255"));
}

void loop() {
  const int raw = analogRead(POT_PIN);
  const int pwm = map(raw, 0, 1023, 0, 255);
  analogWrite(LED_PIN, pwm);

  const unsigned long now = millis();
  if (now - lastReport >= 100) {
    lastReport = now;
    Serial.print(F("ADC:")); Serial.print(raw);
    Serial.print(F(" PWM:")); Serial.print(pwm);
    Serial.print(F(" Voltage:")); Serial.println(raw * (5.0 / 1023.0), 2);
  }
}

실습5) FND 세그먼트
(강사님코드)
// 실습 3: 버튼을 누를 때마다 한 자리 7세그먼트의 숫자 증가
// 공통 캐소드(COM = GND), A~G 각각에 330Ω 저항을 사용합니다.
const byte SEGMENT_PINS[] = {2, 3, 4, 5, 6, 7, 8};  // A, B, C, D, E, F, G
const byte BUTTON_PIN = 10;
const byte LED_PIN = 11;
const byte DIGITS[] = {
  0x3F, 0x06, 0x5B, 0x4F, 0x66,
  0x6D, 0x7D, 0x07, 0x7F, 0x6F
};
byte number = 0;
int lastReading = HIGH;
int stableButton = HIGH;
unsigned long changedAt = 0;

void showNumber() {
  for (byte segment = 0; segment < 7; segment++) {
    digitalWrite(SEGMENT_PINS[segment], bitRead(DIGITS[number], segment));
  }
  digitalWrite(LED_PIN, number == 0 ? HIGH : LOW);
  Serial.print(F("Digit:")); Serial.println(number);
}

void setup() {
  Serial.begin(9600);
  for (byte pin : SEGMENT_PINS) pinMode(pin, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(LED_PIN, OUTPUT);
  Serial.println(F("Counter ready: press button once per number"));
  showNumber();
}

void loop() {
  const int reading = digitalRead(BUTTON_PIN);
  const unsigned long now = millis();
  if (reading != lastReading) changedAt = now;
  lastReading = reading;
  if (now - changedAt >= 35 && reading != stableButton) {
    stableButton = reading;
    if (stableButton == LOW) {
      number = (number + 1) % 10;
      showNumber();
    }
  }
}

실습6) 세그먼트 응용
사용품 : led 3개(빨강, 주황, 초록), 세그먼트, 스위치

문제
내부(아두이노)에서 10(임의로 변경가능)초 카운트 다운을 시작하며, 카운트 다운에따라 빨, 주, 초 순서대로 바뀐다.
2초부터는 초록색이 점등되어야하며, 초록색일때 스위치를 누른다
초록색일때 스위치를 누를시 세그먼트에 +1이 되고, 그외의 상태일때 -1을하여 세그먼트에 표기를 한다.
총 9점이 되었을때 모든 led가 반짝거리며 3초후에 초기화가 된다.
0점 이하로 떨어지는 경우가 없어야한다.
